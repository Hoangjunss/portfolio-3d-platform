- [ ] **Step 10: `templates/wedding/app/computeCountdown.ts`**

```typescript
export interface CountdownParts {
  days: number;
  hours: number;
  minutes: number;
  seconds: number;
  hasPassed: boolean;
}

export function computeCountdown(targetIso: string, now: Date): CountdownParts {
  const targetMs = new Date(targetIso).getTime();
  const diffMs = targetMs - now.getTime();
  if (diffMs <= 0) {
    return { days: 0, hours: 0, minutes: 0, seconds: 0, hasPassed: true };
  }
  const days = Math.floor(diffMs / (1000 * 60 * 60 * 24));
  const hours = Math.floor((diffMs / (1000 * 60 * 60)) % 24);
  const minutes = Math.floor((diffMs / (1000 * 60)) % 60);
  const seconds = Math.floor((diffMs / 1000) % 60);
  return { days, hours, minutes, seconds, hasPassed: false };
}
```

```tsx
'use client';
import { useEffect, useState } from 'react';
import { computeCountdown, type CountdownParts } from './computeCountdown';
import { WEDDING } from '../data/seed';

export function WeddingCountdown() {
  const [parts, setParts] = useState<CountdownParts>(() =>
    computeCountdown(WEDDING.weddingDateIso, new Date()),
  );

  useEffect(() => {
    const id = setInterval(() => {
      setParts(computeCountdown(WEDDING.weddingDateIso, new Date()));
    }, 1000);
    return () => clearInterval(id);
  }, []);

  if (parts.hasPassed) {
    return <p className="wedding-countdown wedding-countdown-passed">We're married!</p>;
  }

  return (
    <div className="wedding-countdown" role="timer" aria-live="polite">
      <div><span>{parts.days}</span><small>days</small></div>
      <div><span>{parts.hours}</span><small>hours</small></div>
      <div><span>{parts.minutes}</span><small>minutes</small></div>
      <div><span>{parts.seconds}</span><small>seconds</small></div>
    </div>
  );
}
```

### Design decisions

**(e) `RsvpSection` is a client component (`'use client'`)** mounted inside the otherwise
static-exported `page.tsx`, at the `id="rsvp"` anchor `Hero`'s CTA scrolls to — per the Global
Constraints render-pattern rule, it owns the `useLocalCollection` call, renders `InquiryForm`, and
derives + renders the live "X guests confirmed" total; the rest of the page (`Hero`,
`WeddingCountdown`, `PhotoGallery`, `Footer`) stays server/static-rendered or, for the countdown,
client-rendered but independent of this hook entirely (Task 2, §4 of the site spec).

**(f) `attending`/`guestCount` normalization happens inside `RsvpSection`'s `onSubmit` handler, not
inside `InquiryForm`** — per `docs/superpowers/specs/2026-09-20-site-wedding-design.md` §5.2/§5.3:
`InquiryForm`'s `fields` prop only supports `type: 'text' | 'email' | 'tel' | 'number'`, so
"attending yes/no" is collected as free-text and parsed here (lowercase/trim, default to `'no'` on
any ambiguous value; `guestCount` clamped to `0` when not attending, minimum `1` when attending).
This keeps `InquiryForm` itself unchanged, matching the "shared behavior, bespoke visual/content
design per site" rule (master spec §5) the same way corporate's `CallbackSection` and realestate's
save-toggle keep their kit components untouched.

**(g) The live guest-count total is derived on every render from `items`, never a separately stored
counter** — per site spec §5.3, `confirmedGuests` is computed with a `filter`+`reduce` over the
hook's `items` array inside the component body, so it can never drift out of sync with the
underlying collection (no second piece of state to keep consistent).

```tsx
'use client';
import { useLocalCollection, InquiryForm } from '@portfolio/template-kit';

interface RsvpEntry {
  id: string;
  name: string;
  attending: 'yes' | 'no';
  guestCount: number;
}

const RSVP_SEED: RsvpEntry[] = [];

const RSVP_EXTRA_FIELDS = [
  { name: 'attending', label: 'Attending? (type "yes" or "no")', type: 'text' as const, required: true },
  { name: 'guestCount', label: 'Number of guests (including yourself)', type: 'number' as const, required: true },
];

export function RsvpSection() {
  const { items, add, reset } = useLocalCollection<RsvpEntry>('wedding-rsvps', RSVP_SEED);

  async function handleSubmit(values: Record<string, string>) {
    const attending: 'yes' | 'no' = values.attending.trim().toLowerCase() === 'yes' ? 'yes' : 'no';
    const rawCount = Number(values.guestCount) || 0;
    const guestCount = attending === 'no' ? 0 : Math.max(1, rawCount);

    add({
      id: crypto.randomUUID(),
      name: values.name,
      attending,
      guestCount,
    });
  }

  const confirmedGuests = items
    .filter((entry) => entry.attending === 'yes')
    .reduce((sum, entry) => sum + entry.guestCount, 0);

  return (
    <section id="rsvp" className="wedding-rsvp">
      <h2>RSVP</h2>
      <p aria-live="polite">{confirmedGuests} guests confirmed</p>
      <InquiryForm fields={RSVP_EXTRA_FIELDS} submitLabel="Send RSVP" onSubmit={handleSubmit} />
      <button type="button" onClick={reset}>Reset demo data</button>
    </section>
  );
}
```
